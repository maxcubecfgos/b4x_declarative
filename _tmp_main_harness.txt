Private Sub RunLifecycleHarness
	Dim lifecycleInput As UIInput
	lifecycleInput.Initialize.Text("lifecycle")
	lifecycleInput.SetParent(Root)
	lifecycleInput.SetPosition(0, 0)
	lifecycleInput.SetSize(1dip, 1dip)
	lifecycleInput.Render
	Dim firstView As B4XView = lifecycleInput.GetView
	Dim firstId As Int = NativeViewIdentity(firstView)

	lifecycleInput.Render
	Dim stableView As B4XView = lifecycleInput.GetView
	Dim stableId As Int = NativeViewIdentity(stableView)
	Dim stableText As String = lifecycleInput.GetText
	If firstId > 0 And firstId = stableId And stableText = "lifecycle" Then
		Log("DeclarativeUI TEST PASS  lifecycle stable render preserves native view")
	Else
		Log("DeclarativeUI TEST FAIL  lifecycle stable render")
	End If

	lifecycleInput.Detach
	lifecycleInput.SetParent(Root)
	lifecycleInput.SetPosition(0, 0)
	lifecycleInput.SetSize(1dip, 1dip)
	lifecycleInput.Render
	Dim remountedView As B4XView = lifecycleInput.GetView
	Dim remountedId As Int = NativeViewIdentity(remountedView)
	If remountedId > 0 And remountedId = firstId And lifecycleInput.GetText = "lifecycle" Then
		Log("DeclarativeUI TEST PASS  lifecycle detach preserves native view and state")
	Else
		Log("DeclarativeUI TEST FAIL  lifecycle detach/remount")
	End If

	lifecycleInput.Unmount
	If lifecycleInput.GetView = Null Then
		Log("DeclarativeUI TEST PASS  lifecycle unmount releases native view")
	Else
		Log("DeclarativeUI TEST FAIL  lifecycle unmount release")
	End If

	RunNavigatorLifecycleHarness
	RunListLifecycleHarness
End Sub

Private Sub RunNavigatorLifecycleHarness
	Dim routeHostPanel As Panel
	routeHostPanel.Initialize("")
	Dim routeHost As B4XView = routeHostPanel
	Root.AddView(routeHost, 0, 0, 1dip, 1dip)

	Dim routeInput As UIInput
	routeInput.Initialize.Text("route-a")
	Dim routeBodyA As UIColumn
	routeBodyA.Initialize.AddChild(routeInput)
	Dim routeA As UIScaffold
	routeA.Initialize.Body(routeBodyA)

	Dim routeLabel As UILabel
	routeLabel.Initialize.Text("route-b")
	Dim routeBodyB As UIColumn
	routeBodyB.Initialize.AddChild(routeLabel)
	Dim routeB As UIScaffold
	routeB.Initialize.Body(routeBodyB)

	Dim routeNavigator As UINavigator
	routeNavigator.Initialize.AddScreen("A", routeA).AddScreen("B", routeB)
	routeNavigator.SetParent(routeHost)
	routeNavigator.SetPosition(0, 0)
	routeNavigator.SetSize(1dip, 1dip)
	routeNavigator.Render
	Dim initialText As String = routeInput.GetText
	routeNavigator.NavigateTo("B")
	Dim openedBackStack As Boolean = routeNavigator.CanGoBack
	Dim popped As Boolean = routeNavigator.GoBack
	Dim restoredText As String = routeInput.GetText
	If initialText = "route-a" And openedBackStack And popped And restoredText = "route-a" Then
		Log("DeclarativeUI TEST PASS  navigator detach A->B->GoBack preserves route state")
	Else
		Log("DeclarativeUI TEST FAIL  navigator route detach/restore")
	End If
	routeNavigator.Unmount
	routeHost.RemoveViewFromParent
End Sub

Private Sub RunListLifecycleHarness
	LifecycleCreateCount = 0
	LifecycleBindCount = 0
	Dim data As List
	data.Initialize
	For i = 0 To 49
		data.Add("item-" & i)
	Next
	LifecycleList.Initialize.Items(data).ItemHeight(40dip).Overscan(0) _
		.CreateItem(Me, "HarnessCreateItem").BindItem(Me, "HarnessBindItem")
	LifecycleList.SetParent(Root)
	LifecycleList.SetPosition(0, 0)
	LifecycleList.SetSize(240dip, 120dip)
	LifecycleList.Render
	Dim baselineCreate As Int = LifecycleCreateCount
	Dim baselineBind As Int = LifecycleBindCount
	Dim nativeList As B4XView = LifecycleList.GetView
	Dim scroll As ScrollView = nativeList
	scroll.ScrollPosition = 800dip
	LifecycleList.Render
	Dim recycledCreate As Int = LifecycleCreateCount
	Dim recycledBind As Int = LifecycleBindCount
	If baselineCreate > 0 And recycledCreate = baselineCreate And recycledBind > baselineBind Then
		Log("DeclarativeUI TEST PASS  list detach pool reuses rows")
		Log("DeclarativeUI TEST INFO  list create=" & recycledCreate & " bind=" & recycledBind)
	Else
		Log("DeclarativeUI TEST FAIL  list detach pool reuse")
		Log("DeclarativeUI TEST INFO  list create=" & recycledCreate & " bind=" & recycledBind)
	End If
	LifecycleList.Unmount
End Sub

Private Sub NativeViewIdentity(View As B4XView) As Int
	If View = Null Then Return 0
	If View.IsInitialized = False Then Return 0
	Dim system As JavaObject
	system.InitializeStatic("java.lang.System")
	Return system.RunMethod("identityHashCode", Array(View))
End Sub

Private Sub HarnessCreateItem(Index As Int) As Object
	LifecycleCreateCount = LifecycleCreateCount + 1
	Dim label As UILabel
	label.Initialize.Size(12)
	Return label
End Sub

Private Sub HarnessBindItem(Index As Int, ItemView As Object)
	LifecycleBindCount = LifecycleBindCount + 1
	Dim label As UILabel = ItemView
	label.Text("item-" & Index)
End Sub
